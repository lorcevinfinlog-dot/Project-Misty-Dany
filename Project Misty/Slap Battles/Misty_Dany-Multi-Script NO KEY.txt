local function playerColor(p)
   if p.Team and lp.Team then
      return p.Team == lp.Team and Color3.fromRGB(60,220,90) or Color3.fromRGB(255,60,60)
   end
   return Color3.fromRGB(255,195,0)
end

local function worldToScreen(pos)

end

local function getCharAABB(char)
   local mn = Vector3.new(math.huge, math.huge, math.huge)
   local mx = Vector3.new(-math.huge,-math.huge,-math.huge)
   for _, p in ipairs(char:GetDescendants()) do
      if p:IsA("BasePart") and p.Name ~= "HumanoidRootPart" then
         local h = p.Size / 2
         mn = Vector3.new(math.min(mn.X, p.Position.X-h.X), math.min(mn.Y, p.Position.Y-h.Y), math.min(mn.Z, p.Position.Z-h.Z))
         mx = Vector3.new(math.max(mx.X, p.Position.X+h.X), math.max(mx.Y, p.Position.Y+h.Y), math.max(mx.Z, p.Position.Z+h.Z))
      end
   end
   return mn, mx
end

local function buildESP(player)
   if player == lp or espObjects[player] then return end

   -- Highlight (сквозь стены)
   local hl = Instance.new("Highlight")
   hl.DepthMode           = Enum.HighlightDepthMode.AlwaysOnTop
   hl.FillTransparency    = 0.5
   hl.OutlineTransparency = 0
   hl.Enabled             = false

   -- Billboard
   local bb = Instance.new("BillboardGui")
   bb.AlwaysOnTop  = true
   bb.Size         = UDim2.new(0, 200, 0, 70)
   bb.StudsOffset  = Vector3.new(0, 3.5, 0)
   bb.Enabled      = false

   -- Имя
   local nameLbl = Instance.new("TextLabel", bb)
   nameLbl.Size = UDim2.new(1,0,0,22)
   nameLbl.Position = UDim2.new(0,0,0,0)
   nameLbl.BackgroundTransparency = 1
   nameLbl.Font = Enum.Font.GothamBold
   nameLbl.TextSize = 14
   nameLbl.TextStrokeTransparency = 0.3
   nameLbl.TextStrokeColor3 = Color3.new(0,0,0)

   -- HP bar фон
   local hpBg = Instance.new("Frame", bb)
   hpBg.Size = UDim2.new(1,0,0,6)
   hpBg.Position = UDim2.new(0,0,0,24)
   hpBg.BackgroundColor3 = Color3.fromRGB(30,30,30)
   hpBg.BorderSizePixel = 0
   Instance.new("UICorner", hpBg).CornerRadius = UDim.new(1,0)

   local hpBar = Instance.new("Frame", hpBg)
   hpBar.Size = UDim2.new(1,0,1,0)
   hpBar.BackgroundColor3 = Color3.fromRGB(80,220,80)
   hpBar.BorderSizePixel = 0
   Instance.new("UICorner", hpBar).CornerRadius = UDim.new(1,0)

   -- Дистанция
   local distLbl = Instance.new("TextLabel", bb)
   distLbl.Size = UDim2.new(1,0,0,16)
   distLbl.Position = UDim2.new(0,0,0,34)
   distLbl.BackgroundTransparency = 1
   distLbl.Font = Enum.Font.Gotham
   distLbl.TextSize = 12
   distLbl.TextColor3 = Color3.fromRGB(200,200,200)
   distLbl.TextStrokeTransparency = 0.4
   distLbl.TextStrokeColor3 = Color3.new(0,0,0)

   -- Drawing линия (правильный способ для ESP линий)
   local lineD = Drawing.new("Line")
   lineD.Thickness = 1.5
   lineD.Color     = Color3.fromRGB(255,195,0)
   lineD.Visible   = false
   lineD.ZIndex    = 5

   -- Drawing боксы (4 линии = прямоугольник)
   local boxLines = {}
   for i = 1,4 do
      local bl = Drawing.new("Line")
      bl.Thickness = 1.5
      bl.Color     = Color3.fromRGB(255,195,0)
      bl.Visible   = false
      bl.ZIndex    = 5
      boxLines[i]  = bl
   end

   espObjects[player] = {
      hl=hl, bb=bb, nameLbl=nameLbl, hpBar=hpBar, hpBg=hpBg,
      distLbl=distLbl,
      lineD=lineD,
      boxLines=boxLines,
   }

   local function attach(char)
      local obj = espObjects[player]
      if not obj then return end
      obj.hl.Parent = char
      local hrp = char:WaitForChild("HumanoidRootPart", 5)
      if hrp then
         obj.bb.Adornee = hrp
         obj.bb.Parent  = hrp
      end
   end

   if player.Character then attach(player.Character) end
   player.CharacterAdded:Connect(attach)
end

local function removeESP(player)
   local o = espObjects[player]
   if not o then return end
   pcall(function() o.hl:Destroy() end)
   pcall(function() o.bb:Destroy() end)
   pcall(function() o.lineD:Remove() end)
   for _,bl in ipairs(o.boxLines or {}) do pcall(function() bl:Remove() end) end
   espObjects[player] = nil
end

-- Drawing API — единственный правильный способ рисовать линии ESP
local function setDrawLine(dl, a, b, col)
   dl.From    = a
   dl.To      = b
   dl.Color   = col
   dl.Visible = true
end

-- box: TL, TR, BR, BL → 4 линии
local function setDrawBox(bls, tl, br, col)
   local tr = Vector2.new(br.X, tl.Y)
   local bl = Vector2.new(tl.X, br.Y)
   setDrawLine(bls[1], tl, tr, col)  -- верх
   setDrawLine(bls[2], bl, br, col)  -- низ
   setDrawLine(bls[3], tl, bl, col)  -- левый
   setDrawLine(bls[4], tr, br, col)  -- правый
end

local function startESP()
   if espConn then espConn:Disconnect() end
   espConn = RunService.RenderStepped:Connect(function()
      if not espEnabled then return end
      local myHRP = getHRP()
      local vp = cam.ViewportSize
      -- Линия от моего персонажа (ноги), проецированного на экран
      -- Если персонаж не виден — fallback на низ экрана
      local bot
      if myHRP then
         local myFeet = myHRP.Position - Vector3.new(0, 3, 0)
         local s, onS = cam:WorldToViewportPoint(myFeet)
         if onS and s.Z > 0 then
            bot = Vector2.new(s.X, s.Y)
         else
            bot = Vector2.new(vp.X / 2, vp.Y)
         end
      else
         bot = Vector2.new(vp.X / 2, vp.Y)
      end

      for player, o in pairs(espObjects) do
         local char = player.Character
         local hrp  = char and char:FindFirstChild("HumanoidRootPart")
         local hum  = char and char:FindFirstChildOfClass("Humanoid")

         if hrp and hum and hum.Health > 0 then
            local col = playerColor(player)
            local dist = myHRP and math.floor((myHRP.Position-hrp.Position).Magnitude) or 0
            local hpPct = hum.Health / math.max(hum.MaxHealth, 1)

            -- Цвет HP бара (зелёный → красный)
            o.hpBar.BackgroundColor3 = Color3.fromRGB(
               math.floor((1-hpPct)*220),
               math.floor(hpPct*220),
               50
            )
            o.hpBar.Size = UDim2.new(hpPct,0,1,0)

            -- Highlight
            o.hl.Enabled = espEnabled and espHighlight
            o.hl.FillColor = col
            o.hl.OutlineColor = Color3.new(1,1,1)

            -- Billboard
            o.bb.Enabled = espEnabled and espNames
            o.nameLbl.Text = player.DisplayName
            o.nameLbl.TextColor3 = col
            o.hpBg.Visible = espHealth
            o.distLbl.Text = espDist and (tostring(dist).." м") or ""
            o.distLbl.Visible = espDist

            -- Линия (Drawing API — точная, без багов с rotation)
            local sc, onS = worldToScreen(hrp.Position)
            if espLines and onS then
               setDrawLine(o.lineD, bot, sc, col)
            else
               o.lineD.Visible = false
            end

            -- Бокс (Drawing API — 4 линии по углам хитбокса)
            if espBoxes then
               local mn, mx = getCharAABB(char)
               -- Проецируем только нужные 4 угла (top-left, top-right, bot-left, bot-right)
               local pTL, onTL = worldToScreen(Vector3.new(mn.X, mx.Y, mn.Z))
               local pTR, onTR = worldToScreen(Vector3.new(mx.X, mx.Y, mn.Z))
               local pBL, onBL = worldToScreen(Vector3.new(mn.X, mn.Y, mn.Z))
               local pBR, onBR = worldToScreen(Vector3.new(mx.X, mn.Y, mn.Z))
               -- Берём крайние X/Y чтобы бокс не зависел от угла камеры
               local scMinX = math.min(pTL.X, pTR.X, pBL.X, pBR.X)
               local scMinY = math.min(pTL.Y, pTR.Y, pBL.Y, pBR.Y)
               local scMaxX = math.max(pTL.X, pTR.X, pBL.X, pBR.X)
               local scMaxY = math.max(pTL.Y, pTR.Y, pBL.Y, pBR.Y)
               if onS then
                  setDrawBox(o.boxLines, Vector2.new(scMinX, scMinY), Vector2.new(scMaxX, scMaxY), col)
               else
                  for _,bl in ipairs(o.boxLines) do bl.Visible = false end
               end
            else
               for _,bl in ipairs(o.boxLines) do bl.Visible = false end
            end
         else
            -- Персонаж мёртв или не загружен — скрываем всё
            o.hl.Enabled = false
            o.bb.Enabled = false
            o.lineD.Visible = false
            for _,bl in ipairs(o.boxLines) do bl.Visible = false end
         end
      end
   end)
end

local function stopESP()
   if espConn then espConn:Disconnect(); espConn=nil end
   for _,o in pairs(espObjects) do
      o.hl.Enabled = false
      o.bb.Enabled = false
      o.lineD.Visible = false
      for _,bl in ipairs(o.boxLines or {}) do bl.Visible = false end
   end
end

-- ────────────────────────────────────────────────────────────────
--  GOD MODE v2  —  правильная реализация
--
--  Проблемы старой версии:
--   1. BodyPosition с MaxForce Y телепортировал персонажа вниз
--   2. MaxHealth=1e9 конфликтовал с серверными скриптами которые
--      сбрасывают HP обратно — это вызывало мерцание и смерть
--
--  Новый подход:
--   • Слушаем событие HealthChanged и мгновенно возвращаем HP
--   • ForceField блокирует урон на уровне движка (не HP напрямую)
--   • Чистим дебаффы (Plague/Burn/Frozen) каждые 0.05 сек
--   • НЕ двигаем HRP — персонаж стоит на месте как обычно
--   • НЕ меняем MaxHealth — не конфликтуем с сервером
-- ────────────────────────────────────────────────────────────────
local godHealthConn  = nil   -- слушатель HP
local godDebuffConn  = nil   -- чистка дебаффов
local godFF          = nil   -- ForceField

local function godModeOn()
   local char = getChar()
   if not char then return end
   local hum  = char:FindFirstChildOfClass("Humanoid")
   local hrp  = char:FindFirstChild("HumanoidRootPart")
   if not hum or not hrp then return end

   -- 1. ForceField — движок не даёт нанести урон
   if not char:FindFirstChildOfClass("ForceField") then
      godFF = Instance.new("ForceField", char)
      godFF.Visible = false
   end

   -- 2. Слушаем HealthChanged — если HP упало, мгновенно ставим обратно
   --    Это работает даже если сервер сбрасывает MaxHealth
   local lastHP = hum.Health
   godHealthConn = hum.HealthChanged:Connect(function(hp)
      if not _G.GodMode then return end
      if hp < lastHP then
         -- Урон прошёл мимо ForceField (серверный скрипт) — восстанавливаем
         hum.Health = hum.MaxHealth
      end
      lastHP = hum.Health
   end)

   -- 3. Чистка дебаффов каждые 0.05 сек
   local DEBUFFS = {
      "Plague","Infected","Burn","Frozen","Stun",
      "Slow","Poison","Bleeding","Ragdolled",
   }
   godDebuffConn = RunService.Heartbeat:Connect(function()
      if not _G.GodMode then return end
      local c = getChar()
      if not c then return end
      for _, name in ipairs(DEBUFFS) do
         local d = c:FindFirstChild(name)
         -- Удаляем Script/LocalScript дебаффов
         -- BoolValue "Ragdolled" только сбрасываем в false, не удаляем
         if d then
            if d:IsA("BoolValue") then
               d.Value = false
            elseif d:IsA("Script") or d:IsA("LocalScript")
               or d:IsA("ModuleScript") then
               pcall(function() d:Destroy() end)
            else
               pcall(function() d:Destroy() end)
            end
         end
      end
      -- Держим ForceField живым (некоторые игры удаляют его)
      if not c:FindFirstChildOfClass("ForceField") then
         godFF = Instance.new("ForceField", c)
         godFF.Visible = false
      end
   end)

   notify("☠️ God Mode", "ВКЛ — ForceField + HP restore", 3)
end

local function godModeOff()
   -- Отключаем listeners
   if godHealthConn then godHealthConn:Disconnect(); godHealthConn = nil end
   if godDebuffConn then godDebuffConn:Disconnect(); godDebuffConn = nil end
   -- Убираем ForceField
   local char = getChar()
   if char then
      for _, ff in pairs(char:GetChildren()) do
         if ff:IsA("ForceField") then ff:Destroy() end
      end
   end
   godFF = nil
   notify("☠️ God Mode", "ВЫКЛ", 2)
end

      godModeOn()
   end
end)

-- Anti-Ragdoll заменён на Ragdoll Bypass через принудительный сброс состояния
local AntiRagToggle = TabPlayer:CreateToggle({
   Name = "🔄 Обход регдола", CurrentValue=false,
   Callback = function(v)
      _G.AntiRagdoll = v
      if v then
         _G.ARC = RunService.Heartbeat:Connect(function()
            if not _G.AntiRagdoll then
               _G.ARC:Disconnect(); return
            end
            local char = getChar(); if not char then return end
            local hum  = char:FindFirstChildOfClass("Humanoid")
            if not hum then return end
            local root = char:FindFirstChild("HumanoidRootPart")

            -- Убираем все BallSocketConstraint / HingeConstraint которые делают регдол
            for _, v2 in pairs(char:GetDescendants()) do
               if v2:IsA("BallSocketConstraint") or v2:IsA("HingeConstraint") then
                  v2.Enabled = false
               end
            end

            -- Сбрасываем состояние регдола
            local state = hum:GetState()
            if state == Enum.HumanoidStateType.Ragdoll
            or state == Enum.HumanoidStateType.FallingDown then
               hum.PlatformStand = false
               hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            end

            -- Удаляем BoolValue "Ragdolled" если игра его юзает
            local rag = char:FindFirstChild("Ragdolled")
            if rag and rag.Value == true then
               rag.Value = false
            end
         end)
      else
         if _G.ARC then _G.ARC:Disconnect() end
         -- Возвращаем constraints
         local char = getChar()
         if char then
            for _, v2 in pairs(char:GetDescendants()) do
               if v2:IsA("BallSocketConstraint") or v2:IsA("HingeConstraint") then
                  v2.Enabled = true
               end
            end
         end
      end
   end
})

local function parseOpt(opt)
   if type(opt) == "table" then
      return tostring(opt[1] or "")

end

local function findPlayerByName(name)
   if not name or name == "" then return nil end

   return nil
end

end

-- ────────────────────────────────────────────────────────────────
--  FPS BOOSTER SYSTEM
--  Принцип: убираем всё что жрёт GPU/CPU не влияя на геймплей.
--  Каждый шаг обратим — при выключении возвращаем оригинал.
-- ────────────────────────────────────────────────────────────────
local FPSBooster = {}
FPSBooster.active      = false
FPSBooster.origShadows = Lighting.GlobalShadows
FPSBooster.origQuality = settings().Rendering.QualityLevel
FPSBooster.origMaxFPS  = settings().Rendering.FrameRateManager
FPSBooster.removedObjs = {}   -- храним удалённые объекты чтобы вернуть

function FPSBooster:Enable()
   if self.active then return end
   self.active = true

   -- 1. Максимальная частота кадров через setfpscap если есть
   if setfpscap then setfpscap(0) end  -- 0 = без ограничений

   -- 2. Отключаем тени (самый большой буст)
   self.origShadows      = Lighting.GlobalShadows
   Lighting.GlobalShadows = false

   -- 3. Убираем туман
   self.origFogEnd   = Lighting.FogEnd
   self.origFogStart = Lighting.FogStart
   Lighting.FogEnd   = 1e6
   Lighting.FogStart = 1e6

   -- 4. Минимальная атмосфера
   for _, v in pairs(Lighting:GetChildren()) do
      if v:IsA("Atmosphere") or v:IsA("BlurEffect")
      or v:IsA("DepthOfFieldEffect") or v:IsA("SunRaysEffect")
      or v:IsA("ColorCorrectionEffect") or v:IsA("BloomEffect") then
         v.Enabled = false
      end
   end

   -- 5. Отключаем все ParticleEmitter / Smoke / Fire в workspace
   for _, v in pairs(workspace:GetDescendants()) do
      if v:IsA("ParticleEmitter") or v:IsA("Smoke")
      or v:IsA("Fire") or v:IsA("Sparkles") or v:IsA("Trail") then
         if v.Enabled then
            v.Enabled = false
            table.insert(self.removedObjs, v)
         end
      end
   end

   -- 6. Понижаем качество рендера если API доступен
   pcall(function()
      settings().Rendering.QualityLevel = 1  -- минимум
   end)

   -- 7. Убираем Decal / Texture со всех Part (жрут VRAM)
   for _, v in pairs(workspace:GetDescendants()) do
      if v:IsA("Decal") or v:IsA("Texture") then
         pcall(function()
            v.Transparency = 1   -- прячем, не удаляем (чтобы вернуть)
            table.insert(self.removedObjs, {obj=v, prop="Transparency", orig=0})
         end)
      end
   end

   notify("⚡ FPS Booster", "Включён — тени/частицы/туман убраны", 3)
end

function FPSBooster:Disable()
   if not self.active then return end
   self.active = false

   -- Возвращаем тени
   Lighting.GlobalShadows = self.origShadows
   Lighting.FogEnd        = self.origFogEnd   or origFogEnd
   Lighting.FogStart      = self.origFogStart or 0

   -- Возвращаем атмосферу
   for _, v in pairs(Lighting:GetChildren()) do
      if v:IsA("Atmosphere") or v:IsA("BlurEffect")
      or v:IsA("DepthOfFieldEffect") or v:IsA("SunRaysEffect")
      or v:IsA("ColorCorrectionEffect") or v:IsA("BloomEffect") then
         v.Enabled = true
      end
   end

   -- Возвращаем частицы
   for _, v in pairs(self.removedObjs) do
      pcall(function()
         if type(v) == "table" then
            v.obj[v.prop] = v.orig
         else
            v.Enabled = true
         end
      end)
   end
   self.removedObjs = {}

   -- Возвращаем качество
   pcall(function()
      settings().Rendering.QualityLevel = self.origQuality
   end)

   notify("⚡ FPS Booster", "Выключён — настройки восстановлены", 3)
end

-- Счётчик FPS через замер времени между кадрами
local fpsSamples    = {}
local fpsMaxSamples = 20   -- усредняем по 20 кадрам

local function getAverageFPS()
   local now = tick()
   table.insert(fpsSamples, now)
   -- Удаляем старые сэмплы (старше 1 сек)
   while fpsSamples[1] and (now - fpsSamples[1]) > 1 do
      table.remove(fpsSamples, 1)
   end
   return #fpsSamples
end

local function getPing()
   -- Стандартный способ получить пинг через Stats
   local ok, v = pcall(function()
      return math.floor(game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue())
   end)
   return ok and tostring(v).." ms" or "N/A"
end

   -- Радужная полоска — плавная смена цвета по времени

-- Функция запуска скрипта с pcall + логом
local function runScript(entry)
   local name   = entry.Name   or "Unknown"
   local source = entry.Source or ""
   local stype  = entry.Type   or "loadstring"

   -- Лог
   local logEntry = {
      Name = name,
      Time = os.date("%H:%M:%S"),
      OK   = false,
      Err  = ""
   }
   table.insert(scriptLog, 1, logEntry)
   if #scriptLog > 30 then table.remove(scriptLog) end

   notify("📝 Скрипт", "Запускаем: " .. name, 2)

   local ok, err = pcall(function()
      if stype == "loadstring" then
         loadstring(game:HttpGet(source, true))()
      elseif stype == "code" then
         loadstring(source)()
      end
   end)

   logEntry.OK  = ok
   logEntry.Err = ok and "" or tostring(err):sub(1, 80)

   if ok then
      notify("✅ Скрипт", name .. " запущен", 2)
   else
      notify("❌ Ошибка", name .. ": " .. logEntry.Err, 5)
   end
end

      local move = Vector3.new()
      if UIS:IsKeyDown(Enum.KeyCode.W) then move += cf.LookVector  end
      if UIS:IsKeyDown(Enum.KeyCode.S) then move -= cf.LookVector  end
      if UIS:IsKeyDown(Enum.KeyCode.A) then move -= cf.RightVector end
      if UIS:IsKeyDown(Enum.KeyCode.D) then move += cf.RightVector end
      if UIS:IsKeyDown(Enum.KeyCode.E) or UIS:IsKeyDown(Enum.KeyCode.Space) then
         move += Vector3.new(0, 1, 0)
      end
      if UIS:IsKeyDown(Enum.KeyCode.Q) or UIS:IsKeyDown(Enum.KeyCode.LeftControl) then
         move -= Vector3.new(0, 1, 0)
      end

      if move.Magnitude > 0 then
         cam.CFrame = cam.CFrame + move.Unit * spd
      end
   end)

   -- Восстанавливаем тип камеры при выключении
   _G.FreecamOrigType = origType
end

   -- 4 линии: верх, низ, лево, право
   for i = 1, 4 do
      local l = Drawing.new("Line")
      l.Color     = crosshairColor
      l.Thickness = crosshairThick
      l.ZIndex    = 20
      l.Visible   = false
      crosshairLines[i] = l

   local dot = Drawing.new("Circle")
   dot.Color   = crosshairColor
   dot.Radius  = 1.5
   dot.Filled  = true
   dot.ZIndex  = 20
   dot.Visible = false
   crosshairLines[5] = dot

   crosshairConn = RunService.RenderStepped:Connect(function()
      if not crosshairEnabled then return end

      -- Низ
      crosshairLines[2].From = Vector2.new(cx, cy + g)
      crosshairLines[2].To   = Vector2.new(cx, cy + g + s)
      -- Лево
      crosshairLines[3].From = Vector2.new(cx - g - s, cy)
      crosshairLines[3].To   = Vector2.new(cx - g,     cy)
      -- Право
      crosshairLines[4].From = Vector2.new(cx + g,     cy)
      crosshairLines[4].To   = Vector2.new(cx + g + s, cy)
      -- Точка
      crosshairLines[5].Position = Vector2.new(cx, cy)

      for _, l in ipairs(crosshairLines) do
         l.Color   = crosshairColor
         l.Visible = true
      end
   end)
end

            freecamLoop = RunService.RenderStepped:Connect(function()
                pcall(function()
                    local currentChar = lp.Character
                    if currentChar and lockedCFrame then
                        local hrp = currentChar:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            -- Намертво пришиваем персонажа к одной точке в пространстве
                            hrp.CFrame = lockedCFrame
                            hrp.AssemblyLinearVelocity = Vector3.zero
                            hrp.AssemblyAngularVelocity = Vector3.zero

                        for _, part in ipairs(currentChar:GetDescendants()) do
                            if part:IsA("BasePart") then

                                part.CanCollide = false
                            end

                        if hum then
                            hum.WalkSpeed = 0
                            hum.JumpPower = 0

                    hum.WalkSpeed = oldWalkSpeed or 16
                    hum.JumpPower = oldJumpPower or 50
                    hum.AutoRotate = oldAutoRotate ~= false
                    hum.PlatformStand = false
                end
            end

            camera.CameraType = oldCameraType or Enum.CameraType.Custom
            camera.CameraSubject = humanoid or oldSubject

            if stopFreecam then pcall(stopFreecam) end
        end
    end
})

-- ────────────────────────────────────────────────────────────────
--  1. CONNECTION MANAGER
--  Все соединения хранятся здесь — один вызов CleanAll() при Unload
-- ────────────────────────────────────────────────────────────────
local ConnectionManager = {}
ConnectionManager._conns = {}   -- { id = RBXScriptConnection }
ConnectionManager._id    = 0

function ConnectionManager:Add(conn, label)
   self._id = self._id + 1
   local id = label or ("conn_" .. self._id)
   self._conns[id] = conn
   return id
end

function ConnectionManager:Remove(id)
   local c = self._conns[id]
   if c then
      pcall(function() c:Disconnect() end)
      self._conns[id] = nil
   end
end

function ConnectionManager:CleanAll()
   for id, c in pairs(self._conns) do
      pcall(function() c:Disconnect() end)
   end
   self._conns = {}
end

function ConnectionManager:Count()
   local n = 0
   for _ in pairs(self._conns) do n = n + 1 end
   return n
end

-- ────────────────────────────────────────────────────────────────
--  2. SETTINGS REGISTRY
--  Сохраняет и загружает все настройки хаба в JSON на диске
-- ────────────────────────────────────────────────────────────────
local SettingsRegistry = {}
SettingsRegistry._data = {}
SettingsRegistry._file = "MistyDany_Settings.json"

function SettingsRegistry:Set(key, value)
   self._data[key] = value
end

function SettingsRegistry:Get(key, default)
   local v = self._data[key]
   return v ~= nil and v or default
end

function SettingsRegistry:SaveAll()
   pcall(function()
      writefile(self._file, HttpService:JSONEncode(self._data))
   end)
end

function SettingsRegistry:LoadAll()
   local ok, data = pcall(function()
      if isfile(self._file) then
         return HttpService:JSONDecode(readfile(self._file))
      end
      return {}
   end)
   if ok and type(data) == "table" then
      self._data = data
   end
end

function ScriptLoader:LoadUserFromDisk()
   local ok, data = pcall(function()

      end
      return {}
   end)
   if ok and type(data) == "table" then

   end

function ScriptLoader:SaveUserToDisk()
   pcall(function()
      writefile(self._userFile, HttpService:JSONEncode(self.userScripts))
   end)
end

function ScriptLoader:AddUserScript(name, source, stype, category, desc)
   -- Проверяем дубликат
   for _, s in ipairs(self.userScripts) do
      if s.Name == name then
         notify("⚠️", "Скрипт '" .. name .. "' уже сохранён", 3)
         return false
      end
   end
   table.insert(self.userScripts, {
      Name     = name,
      Source   = source,
      Type     = stype     or "loadstring",
      Category = category  or "User",
      Desc     = desc      or "",
      Added    = os.date("%d.%m.%Y %H:%M"),
   })
   self:SaveUserToDisk()
   return true
end

function ScriptLoader:RemoveUserScript(name)
   for i, s in ipairs(self.userScripts) do
      if s.Name == name then
         table.remove(self.userScripts, i)
         self:SaveUserToDisk()
         notify("🗑️", "Удалено: " .. name, 2)
         return true
      end
   end
   return false
end

function ScriptLoader:Run(entry)
   if not entry or not entry.Source or entry.Source == "" then
      notify("❌", "Источник пуст", 3); return false
   end
   local name = entry.Name or "Script"
   notify("▶️ Запуск", name, 2)
   local ok, err = pcall(function()
      if entry.Type == "code" then
         loadstring(entry.Source)()
      else
         loadstring(game:HttpGet(entry.Source, true))()
      end
   end)
   if ok then
      notify("✅", name .. " запущен", 2)
      -- Лог
      table.insert(scriptLog, 1, {
         Name = name, Time = os.date("%H:%M:%S"), OK = true, Err = ""
      })
   else
      local e = tostring(err):sub(1, 100)
      notify("❌ Ошибка", name .. ": " .. e, 6)
      table.insert(scriptLog, 1, {
         Name = name, Time = os.date("%H:%M:%S"), OK = false, Err = e
      })
   end
   if #scriptLog > 50 then table.remove(scriptLog) end
   return ok
end

function ScriptLoader:GetAllByCategory()

   local function add(entry)
      local cat = entry.Category or "Other"
      if not cats[cat] then cats[cat] = {} end
      table.insert(cats[cat], entry)
   end
   for _, s in ipairs(self.library)     do add(s) end
   for _, s in ipairs(self.userScripts) do add(s) end
   return cats
end

      if cmd.name:find(query, 1, true) or cmd.desc:lower():find(query, 1, true) then

local PluginSystem = {}
PluginSystem._plugins = {}

function PluginSystem:Register(plugin)
   -- plugin = { Name, Version, Init, Cleanup }
   if not plugin.Name or not plugin.Init then return false end
   if self._plugins[plugin.Name] then
      notify("⚠️ Plugin", plugin.Name .. " уже загружен", 3)
      return false
   end
   local ok, err = pcall(plugin.Init)
   if ok then
      self._plugins[plugin.Name] = plugin
      notify("🔌 Plugin", plugin.Name .. " v" .. (plugin.Version or "?") .. " загружен", 3)
      return true
   else
      notify("❌ Plugin", plugin.Name .. ": " .. tostring(err), 5)
      return false
   end
end

function PluginSystem:Unregister(name)
   local p = self._plugins[name]
   if not p then return end
   if p.Cleanup then pcall(p.Cleanup) end
   self._plugins[name] = nil
   notify("🔌 Plugin", name .. " выгружен", 2)
end

function PluginSystem:UnregisterAll()
   for name, p in pairs(self._plugins) do
      if p.Cleanup then pcall(p.Cleanup) end
   end
   self._plugins = {}
end

function PluginSystem:List()
   local t = {}
   for name, p in pairs(self._plugins) do
      table.insert(t, "• " .. name .. " v" .. (p.Version or "?"))
   end
   return #t > 0 and table.concat(t, "\n") or "Нет плагинов"
end

function ConfigManager:Load(slot)
   slot = slot or self._activeSlot
   local ok, data = pcall(function()
      return HttpService:JSONDecode(readfile(self:_fileName(slot)))
   end)
   if not ok or type(data) ~= "table" then
      notify("❌ Конфиг", "Не найден: " .. slot, 3); return
   end

   -- Применяем загруженные значения
   if data.flySpeed     then flySpeed     = data.flySpeed     end
   if data.speedhackVal then speedhackVal = data.speedhackVal
      local h = getHum(); if h then h.WalkSpeed = speedhackVal end
   end
   if data.freecamSpeed then freecamSpeed = data.freecamSpeed end
   if data.crosshairSize then crosshairSize = data.crosshairSize end
   if data.crosshairGap  then crosshairGap  = data.crosshairGap  end
   if data.espLines     ~= nil then espLines    = data.espLines    end
   if data.espBoxes     ~= nil then espBoxes    = data.espBoxes    end
   if data.espNames     ~= nil then espNames    = data.espNames    end
   if data.espDist      ~= nil then espDist     = data.espDist     end
   if data.espHealth    ~= nil then espHealth   = data.espHealth   end
   if data.espHighlight ~= nil then espHighlight= data.espHighlight end
   if data.theme        then ThemeSystem:Apply(data.theme) end
   if data.userScripts and type(data.userScripts) == "table" then
      ScriptLoader.userScripts = data.userScripts
   end

   self._activeSlot = slot
   notify("📂 Конфиг", "Загружен: " .. slot, 2)
end

function ConfigManager:StartAutoSave(interval)
   interval = interval or 30
   if self._autoConn then self._autoConn:Disconnect() end
   -- Используем task.delay в цикле
   local function loop()
      while true do
         task.wait(interval)
         self:Save()
      end
   end
   task.spawn(loop)
   notify("💾 Авто-сохранение", "Каждые " .. interval .. " сек", 3)
end

-- ────────────────────────────────────────────────────────────────
--  10. DRAGGABLE WATERMARK
--  Перетаскивание водяного знака мышкой
-- ────────────────────────────────────────────────────────────────
local WatermarkDrag = {}
WatermarkDrag.dragging  = false
WatermarkDrag.dragStart = Vector2.new()
WatermarkDrag.startPos  = Vector2.new(10, 10)
WatermarkDrag.pos       = Vector2.new(10, 10)   -- текущая позиция

-- Детектируем клик по области watermark и тащим
ConnectionManager:Add(
   UserInputService.InputBegan:Connect(function(input)
      if input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
      local mp = Vector2.new(input.Position.X, input.Position.Y)
      local p  = WatermarkDrag.pos
      -- Проверяем попадание в bbox watermark (260x28)
      if mp.X >= p.X and mp.X <= p.X + 260
      and mp.Y >= p.Y and mp.Y <= p.Y + 28 then
         WatermarkDrag.dragging  = true
         WatermarkDrag.dragStart = mp
         WatermarkDrag.startPos  = p
      end
   end),
   "watermark_drag_begin"
)

ConnectionManager:Add(
   UserInputService.InputEnded:Connect(function(input)
      if input.UserInputType == Enum.UserInputType.MouseButton1 then
         WatermarkDrag.dragging = false
         -- Сохраняем позицию
         SettingsRegistry:Set("watermark_x", WatermarkDrag.pos.X)
         SettingsRegistry:Set("watermark_y", WatermarkDrag.pos.Y)
      end
   end),
   "watermark_drag_end"
)

ConnectionManager:Add(
   UserInputService.InputChanged:Connect(function(input)
      if not WatermarkDrag.dragging then return end
      if input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
      local mp   = Vector2.new(input.Position.X, input.Position.Y)
      local diff = mp - WatermarkDrag.dragStart
      local newPos = WatermarkDrag.startPos + diff

      -- Клампируем в пределах экрана
      local vp = cam.ViewportSize
      newPos = Vector2.new(
         math.clamp(newPos.X, 0, vp.X - 265),
         math.clamp(newPos.Y, 0, vp.Y - 35)
      )
      WatermarkDrag.pos = newPos

      -- Двигаем все элементы watermark
      pcall(function()
         Watermark.bg.Position  = newPos
         Watermark.bar.Position = newPos
         Watermark.txt.Position = Vector2.new(newPos.X + 6, newPos.Y + 7)
      end)
   end),
   "watermark_drag_move"
)

-- Восстанавливаем позицию из Settings
do
   local sx = SettingsRegistry:Get("watermark_x", 10)
   local sy = SettingsRegistry:Get("watermark_y", 10)
   WatermarkDrag.pos = Vector2.new(sx, sy)
   pcall(function()
      Watermark.bg.Position  = WatermarkDrag.pos
      Watermark.bar.Position = WatermarkDrag.pos
      Watermark.txt.Position = Vector2.new(sx + 6, sy + 7)
   end)
end
